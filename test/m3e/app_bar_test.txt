pub fn app_bar_render_test() {
  let ab = app_bar.new()
  let ab_centered = app_bar.new() |> app_bar.centered(app_bar.IsCentered)
  let ab_for_some = app_bar.new() |> app_bar.for(Some("id"))
  let ab_for_none = app_bar.new() |> app_bar.for(None)
  let ab_size_small = app_bar.new() |> app_bar.size(app_bar_size.Small)
  let ab_size_medium = app_bar.new() |> app_bar.size(app_bar_size.Medium)
  let ab_size_large = app_bar.new() |> app_bar.size(app_bar_size.Large)

  // 1. Define the test cases in a list of tuples: #(input, expected)
  let cases = [
    // Happy path with no attributes nor children
    #(#(ab, [], []), element.element("m3e-app-bar", [], [])),
    // Happy path with no children
    #(
      #(ab, [attribute.id("id")], []),
      element.element("m3e-app-bar", [attribute.id("id")], []),
    ),
    // Happy path with no attributes
    #(
      #(ab, [], [html.br([])]),
      element.element("m3e-app-bar", [], [html.br([])]),
    ),
    // Happy path with a centered attribute
    #(
      #(ab_centered, [], []),
      element.element("m3e-app-bar", [attribute.attribute("centered", "")], []),
    ),
    // Happy path with a for Some attribute
    #(
      #(ab_for_some, [], []),
      element.element("m3e-app-bar", [attribute.attribute("for", "id")], []),
    ),
    // Happy path with a for None attribute
    #(#(ab_for_none, [], []), element.element("m3e-app-bar", [], [])),
    // Happy path with a size attribute
    #(#(ab_size_small, [], []), element.element("m3e-app-bar", [], [])),
    #(
      #(ab_size_medium, [], []),
      element.element(
        "m3e-app-bar",
        [attribute.attribute("size", "medium")],
        [],
      ),
    ),
    #(
      #(ab_size_large, [], []),
      element.element("m3e-app-bar", [attribute.attribute("size", "large")], []),
    ),
  ]

  // 2. Iterate over the list and run assertions for each
  list.each(cases, fn(c) {
    let #(#(ab, attrs, kids), expected) = c

    app_bar.render(ab, attrs, kids)
    |> should.equal(expected)
  })
}
